import { serve } from "https://deno.land/std@0.168.0/http/server.ts"
import { createClient } from "https://esm.sh/@supabase/supabase-js@2"

const TELEGRAM_BOT_TOKEN = Deno.env.get('TELEGRAM_BOT_TOKEN')
const SUPABASE_URL = Deno.env.get('SUPABASE_URL') 
const SUPABASE_PUBLISHABLE = Deno.env.get('SUPABASE_PUBLISHABLE_KEYS') 
const WEBSITE_URL = Deno.env.get('WEBSITE_URL') || 'https://swiftotp.online'

const supabase = createClient(SUPABASE_URL!, SUPABASE_PUBLISHABLE!)

async function sendMessage(chatId: number, text: string) {
  const url = `https://api.telegram.org/bot${TELEGRAM_BOT_TOKEN}/sendMessage`
  await fetch(url, {
    method: 'POST',
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify({ chat_id: chatId, text: text, parse_mode: 'HTML' }),
  })
}

serve(async (req) => {
  try {
    if (req.method !== 'POST') return new Response('Method not allowed', { status: 405 })

    const update = await req.json()

    if (update.message && update.message.text) {
      const chatId = update.message.chat.id
      const text = update.message.text.trim().toLowerCase()
      
      const args = text.split(' ')
      const command = args[0]

      if (command === '/start') {
        await sendMessage(chatId, `?? Welcome to <b>SwiftOTP</b>!\n\nYour Telegram ID is: <code>${chatId}</code>\n\nLink this ID in your Profile settings on the website to use the bot!`)
      } 
      else if (command === '/balance') {
        // SMART ROUTING: Handles simple commands locally via SQL RPC
        const { data: balance, error } = await supabase.rpc('get_balance_by_telegram_id', { 
          tg_id: chatId.toString() 
        })

        if (balance !== null && !error) {
          await sendMessage(chatId, `?? Your wallet balance is: <b>$${Number(balance).toFixed(2)}</b>`)
        } else {
          await sendMessage(chatId, `? Account not linked! Your ID: <code>${chatId}</code>`)
        }
      }
      else if (command === '/buy') {
        // SMART ROUTING: Forward complex core actions to Next.js Vercel backend
        await sendMessage(chatId, `? Processing your order...`)
        
        try {
          const res = await fetch(`${WEBSITE_URL}/api/bot/command`, {
            method: 'POST',
            headers: { 
              'Content-Type': 'application/json',
              'x-telegram-bot-token': TELEGRAM_BOT_TOKEN! 
            },
            body: JSON.stringify({
              telegramId: chatId.toString(),
              command: 'buy',
              args: args.slice(1) // ["google", "usa"]
            })
          })
          
          const data = await res.json()
          await sendMessage(chatId, data.text)
        } catch (e) {
          await sendMessage(chatId, `? Failed to connect to SwiftOTP servers.`)
        }
      }
      else {
        await sendMessage(chatId, `?? <b>SwiftOTP Bot</b>\n\nCommands:\n/balance - Check wallet balance\n/buy - Buy a number (e.g., /buy google usa)`)
      }
    }

    return new Response('OK', { status: 200 })
  } catch (error) {
    console.error(error)
    return new Response('Error', { status: 500 })
  }
})
